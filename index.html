/** * ระบบบันทึกรายรับรายจ่ายประจำวัน - ร้านฟื้นฟูมือถือ * วางโค้ดนี้ใน Google Sheet > ส่วนขยาย > Apps Script */ const SHEET_INCOME = 'บันทึกรายรับ'; const SHEET_EXPENSE = 'บันทึกรายจ่าย'; const SHEET_SUMMARY = 'สรุปรายวัน'; const HEADERS = { [SHEET_INCOME]: [ 'วันที่', 'เลขที่ใบงาน', 'ชื่อลูกค้า', 'เบอร์โทร', 'ยี่ห้อ/รุ่นมือถือ', 'อาการเสีย', 'ประเภทงาน', 'ค่าอะไหล่', 'ค่าแรง', 'ส่วนลด', 'ยอดรวม', 'วิธีชำระเงิน', 'สถานะงาน', 'ผู้รับงาน', 'หมายเหตุ' ], [SHEET_EXPENSE]: [ 'วันที่', 'เลขที่รายการ', 'หมวดรายจ่าย', 'รายละเอียด', 'ร้านค้า/ผู้ขาย', 'จำนวนเงิน', 'วิธีชำระเงิน', 'ผู้บันทึก', 'หมายเหตุ' ], [SHEET_SUMMARY]: [ 'วันที่', 'รายรับรวม', 'รายจ่ายรวม', 'กำไร/ขาดทุน', 'รับเงินสด', 'รับเงินโอน', 'จำนวนงานที่รับ' ] }; const JOB_TYPES = ['เปลี่ยนจอ', 'เปลี่ยนแบตเตอรี่', 'เปลี่ยนช่องชาร์จ', 'ซ่อมบอร์ด', 'ลงโปรแกรม', 'ซ่อมน้ำเข้า', 'อื่นๆ']; const PAYMENT_METHODS = ['เงินสด', 'โอน']; const JOB_STATUS = ['รอซ่อม', 'กำลังซ่อม', 'ซ่อมเสร็จ', 'ส่งมอบแล้ว']; const EXPENSE_CATEGORIES = ['ซื้ออะไหล่', 'ค่าเช่า', 'ค่าไฟ', 'ค่าน้ำ', 'ค่าอินเทอร์เน็ต', 'เงินเดือน', 'อื่นๆ']; /* --------------------------------------------------------- * เมนูและการตั้งค่าเริ่มต้น * ------------------------------------------------------- */ function onOpen() { SpreadsheetApp.getUi() .createMenu('ฟื้นฟูมือถือ') .addItem('ตั้งค่าชีตเริ่มต้น', 'setupSheets') .addItem('อัปเดตสรุปวันนี้', 'updateTodaySummary') .addToUi(); } /** สร้างชีตและหัวข้อทั้งหมด (รันครั้งแรกครั้งเดียว) */ function setupSheets() { const ss = SpreadsheetApp.getActiveSpreadsheet(); Object.keys(HEADERS).forEach(function (name) { let sh = ss.getSheetByName(name); if (!sh) sh = ss.insertSheet(name); const headers = HEADERS[name]; sh.getRange(1, 1, 1, headers.length) .setValues([headers]) .setFontWeight('bold') .setBackground('#d9ead3') .setHorizontalAlignment('center'); sh.setFrozenRows(1); sh.getRange('A2:A').setNumberFormat('dd/mm/yyyy'); }); // ชีตรายรับ const inc = ss.getSheetByName(SHEET_INCOME); inc.getRange('D2:D').setNumberFormat('@'); // เก็บเบอร์โทรเป็นข้อความ จะได้ไม่ตัดเลข 0 inc.getRange('H2:K').setNumberFormat('#,##0.00'); setDropdown_(inc, 'G2:G1000', JOB_TYPES); setDropdown_(inc, 'L2:L1000', PAYMENT_METHODS); setDropdown_(inc, 'M2:M1000', JOB_STATUS); // ชีตรายจ่าย const exp = ss.getSheetByName(SHEET_EXPENSE); exp.getRange('F2:F').setNumberFormat('#,##0.00'); setDropdown_(exp, 'C2:C1000', EXPENSE_CATEGORIES); setDropdown_(exp, 'G2:G1000', PAYMENT_METHODS); // ชีตสรุป const sum = ss.getSheetByName(SHEET_SUMMARY); sum.getRange('B2:F').setNumberFormat('#,##0.00'); // สีสถานะงาน const rules = [ SpreadsheetApp.newConditionalFormatRule() .whenTextEqualTo('ซ่อมเสร็จ').setBackground('#b6d7a8') .setRanges([inc.getRange('M2:M1000')]).build(), SpreadsheetApp.newConditionalFormatRule() .whenTextEqualTo('รอซ่อม').setBackground('#fff2cc') .setRanges([inc.getRange('M2:M1000')]).build() ]; inc.setConditionalFormatRules(rules); } function setDropdown_(sheet, a1, list) { const rule = SpreadsheetApp.newDataValidation() .requireValueInList(list, true) .setAllowInvalid(false) .build(); sheet.getRange(a1).setDataValidation(rule); } /* --------------------------------------------------------- * Web App: รับ-ส่งข้อมูล (ใช้กับแอปหรือฟอร์มภายนอก) * ------------------------------------------------------- */ /** * รับข้อมูลแบบ POST (JSON) * ตัวอย่างบันทึกรายรับ: * {"action":"addIncome","date":"2026-11-17","customer":"คุณสมชาย","phone":"0812345678", * "model":"Samsung A54","problem":"จอแตก","jobType":"เปลี่ยนจอ","parts":1200, * "labor":300,"discount":0,"payment":"เงินสด","status":"รอซ่อม","staff":"ช่างเล็ก","note":""} */ function doPost(e) { const lock = LockService.getScriptLock(); try { lock.waitLock(20000); const data = JSON.parse(e.postData.contents); let result; switch (data.action) { case 'addIncome': result = addIncome_(data); break; case 'addExpense': result = addExpense_(data); break; case 'updateStatus': result = updateJobStatus_(data.jobId, data.status); break; default: throw new Error('ไม่รู้จัก action: ' + data.action); } return jsonOut_({ ok: true, result: result }); } catch (err) { return jsonOut_({ ok: false, error: String(err.message || err) }); } finally { lock.releaseLock(); } } /** * ดึงข้อมูลแบบ GET * ?action=income&date=2026-11-17 * ?action=expense&date=2026-11-17 * ?action=summary&date=2026-11-17 */ function doGet(e) { try { const p = (e && e.parameter) || {}; const date = p.date || ''; let data; switch (p.action) { case 'income': data = getRows_(SHEET_INCOME, date); break; case 'expense': data = getRows_(SHEET_EXPENSE, date); break; case 'summary': data = calcSummary_(date || dateKey_(new Date())); break; default: data = { message: 'ระบบบันทึกรายรับรายจ่ายร้านฟื้นฟูมือถือ ทำงานปกติ' }; } return jsonOut_({ ok: true, data: data }); } catch (err) { return jsonOut_({ ok: false, error: String(err.message || err) }); } } function jsonOut_(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } /* --------------------------------------------------------- * ฟังก์ชันบันทึกข้อมูล * ------------------------------------------------------- */ function addIncome_(d) { const sh = getSheet_(SHEET_INCOME); const row = sh.getLastRow() + 1; const id = nextId_(sh, 'R'); const date = parseDate_(d.date); sh.getRange(row, 1, 1, 15).setValues([[ date, id, d.customer || '', '', // เบอร์โทร (ตั้งค่าแยกด้านล่างเพื่อเก็บเป็นข้อความ) d.model || '', d.problem || '', d.jobType || '', toNumber_(d.parts), toNumber_(d.labor), toNumber_(d.discount), '=H' + row + '+I' + row + '-J' + row, // ยอดรวม d.payment || 'เงินสด', d.status || 'รอซ่อม', d.staff || '', d.note || '' ]]); sh.getRange(row, 4).setNumberFormat('@').setValue(String(d.phone || '')); sh.getRange(row, 1).setNumberFormat('dd/mm/yyyy'); updateSummary_(dateKey_(date)); return { jobId: id, row: row }; } function addExpense_(d) { const sh = getSheet_(SHEET_EXPENSE); const row = sh.getLastRow() + 1; const id = nextId_(sh, 'E'); const date = parseDate_(d.date); sh.getRange(row, 1, 1, 9).setValues([[ date, id, d.category || 'อื่นๆ', d.detail || '', d.vendor || '', toNumber_(d.amount), d.payment || 'เงินสด', d.recorder || '', d.note || '' ]]); sh.getRange(row, 1).setNumberFormat('dd/mm/yyyy'); updateSummary_(dateKey_(date)); return { expenseId: id, row: row }; } /** อัปเดตสถานะงานซ่อมตามเลขที่ใบงาน */ function updateJobStatus_(jobId, status) { if (!jobId || !status) throw new Error('ต้องระบุ jobId และ status'); const sh = getSheet_(SHEET_INCOME); const found = sh.getRange(2, 2, Math.max(sh.getLastRow() - 1, 1), 1) .createTextFinder(jobId).matchEntireCell(true).findNext(); if (!found) throw new Error('ไม่พบใบงาน ' + jobId); sh.getRange(found.getRow(), 13).setValue(status); return { jobId: jobId, status: status }; } /* --------------------------------------------------------- * สรุปรายวัน * ------------------------------------------------------- */ function calcSummary_(key) { const incomeRows = getSheet_(SHEET_INCOME).getDataRange().getValues().slice(1); const expenseRows = getSheet_(SHEET_EXPENSE).getDataRange().getValues().slice(1); let income = 0, expense = 0, cash = 0, transfer = 0, jobs = 0; incomeRows.forEach(function (r) { if (!r[0] || dateKey_(r[0]) !== key) return; const total = toNumber_(r[10]); income += total; jobs += 1; if (r[11] === 'เงินสด') cash += total; if (r[11] === 'โอน') transfer += total; }); expenseRows.forEach(function (r) { if (!r[0] || dateKey_(r[0]) !== key) return; expense += toNumber_(r[5]); }); return { date: key, income: income, expense: expense, profit: income - expense, cash: cash, transfer: transfer, jobs: jobs }; } function updateSummary_(key) { const s = calcSummary_(key); const sh = getSheet_(SHEET_SUMMARY); const dateObj = parseDate_(key); // หาแถวของวันนั้น ถ้าไม่มีให้เพิ่มแถวใหม่ const values = sh.getDataRange().getValues(); let row = -1; for (let i = 1; i < values.length; i++) { if (values[i][0] && dateKey_(values[i][0]) === key) { row = i + 1; break; } } if (row === -1) row = sh.getLastRow() + 1; sh.getRange(row, 1, 1, 7).setValues([[ dateObj, s.income, s.expense, s.profit, s.cash, s.transfer, s.jobs ]]); sh.getRange(row, 1).setNumberFormat('dd/mm/yyyy'); sh.getRange(row, 2, 1, 5).setNumberFormat('#,##0.00'); } function updateTodaySummary() { updateSummary_(dateKey_(new Date())); } /* --------------------------------------------------------- * ฟังก์ชันช่วยเหลือ * ------------------------------------------------------- */ function getSheet_(name) { const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name); if (!sh) throw new Error('ไม่พบชีต "' + name + '" กรุณารัน setupSheets() ก่อน'); return sh; } /** สร้างเลขที่รันนิ่ง เช่น R0001, E0001 */ function nextId_(sheet, prefix) { const last = sheet.getLastRow(); let n = 0; if (last > 1) { n = parseInt(String(sheet.getRange(last, 2).getValue()).replace(/\D/g, ''), 10) || 0; } return prefix + String(n + 1).padStart(4, '0'); } /** แปลงวันที่จาก yyyy-MM-dd หรือ dd/MM/yyyy (รองรับปี พ.ศ.) */ function parseDate_(s) { if (!s) return new Date(); if (s instanceof Date) return s; const str = String(s).trim(); let m = str.match(/^(\d{4})-(\d{1,2})-(\d{1,2})$/); if (m) return new Date(+m[1], +m[2] - 1, +m[3]); m = str.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/); if (m) { let y = +m[3]; if (y > 2400) y -= 543; return new Date(y, +m[2] - 1, +m[1]); } const d = new Date(str); return isNaN(d.getTime()) ? new Date() : d; } function dateKey_(v) { const d = v instanceof Date ? v : parseDate_(v); return Utilities.formatDate(d, Session.getScriptTimeZone(), 'yyyy-MM-dd'); } function toNumber_(v) { const n = Number(String(v === undefined || v === null ? 0 : v).replace(/,/g, '')); return isNaN(n) ? 0 : n; } /** ดึงแถวเป็น object ตามหัวตาราง (กรองตามวันที่ได้) */ function getRows_(sheetName, dateStr) { const values = getSheet_(sheetName).getDataRange().getValues(); const headers = values[0]; const key = dateStr ? dateKey_(dateStr) : ''; return values.slice(1) .filter(function (r) { return r[0] && (!key || dateKey_(r[0]) === key); }) .map(function (r) { const obj = {}; headers.forEach(function (h, i) { obj[h] = r[i] instanceof Date ? dateKey_(r[i]) : r[i]; }); return obj; }); } /* --------------------------------------------------------- * ตัวอย่างสำหรับทดสอบ (เลือกฟังก์ชันแล้วกด "เรียกใช้") * ------------------------------------------------------- */ function testAddIncome() { Logger.log(addIncome_({ date: new Date(), customer: 'คุณสมชาย', phone: '0812345678', model: 'Samsung A54', problem: 'จอแตก', jobType: 'เปลี่ยนจอ', parts: 1200, labor: 300, discount: 0, payment: 'เงินสด', status: 'รอซ่อม', staff: 'ช่างเล็ก', note: 'รับประกัน 30 วัน' })); } function testAddExpense() { Logger.log(addExpense_({ date: new Date(), category: 'ซื้ออะไหล่', detail: 'ซื้อจอ Samsung A54', vendor: 'ร้านอะไหล่ ABC', amount: 800, payment: 'โอน', recorder: 'ช่างเล็ก' }));
